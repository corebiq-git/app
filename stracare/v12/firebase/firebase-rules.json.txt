rules_version = '2';
service cloud.firestore {
match /databases/{database}/documents {

// 1. Helper Functions  
function getUserData() {  
  return get(/databases/$(database)/documents/users/$(request.auth.token.email)).data;  
}  
function isAdmin() {  
  return getUserData().role == 'admin';  
}  
function isUserBranch(docBranch) {  
  return getUserData().branch == docBranch;  
}  

// 2. User Access & Setup Collections  
match /users/{email} {  
  allow read: if request.auth != null && request.auth.token.email == email;  
  allow read, write, delete: if isAdmin();  
}  
match /branches/{document=**} {  
  allow read: if request.auth != null;  
  allow write, delete: if isAdmin();  
}  

// 3. Branch-Restricted Collections (Staff, Appointments, Therapists, etc.)  
match /staff/{document=**} {  
  allow read, write, delete: if isAdmin();  
  allow read: if isUserBranch(resource.data.branch);  
  allow create: if isUserBranch(request.resource.data.branch);  
  allow update: if isUserBranch(resource.data.branch) && isUserBranch(request.resource.data.branch);  
}  
match /appointments/{document=**} {  
  allow read, write, delete: if isAdmin();  
  allow read: if isUserBranch(resource.data.branch);  
  allow create: if isUserBranch(request.resource.data.branch);  
  allow update: if isUserBranch(resource.data.branch) && isUserBranch(request.resource.data.branch);  
}  
match /patients/{document=**} {  
  allow read, write, delete: if isAdmin();  
  allow read: if isUserBranch(resource.data.branch);  
  allow create: if isUserBranch(request.resource.data.branch);  
  allow update: if isUserBranch(resource.data.branch) && isUserBranch(request.resource.data.branch);  
}  
match /transactions/{document=**} {  
  allow read, write, delete: if isAdmin();  
  allow read: if isUserBranch(resource.data.branch);  
  allow create: if isUserBranch(request.resource.data.branch);  
  allow update: if isUserBranch(resource.data.branch) && isUserBranch(request.resource.data.branch);  
}  
match /therapists/{document=**} {  
  allow read, write, delete: if isAdmin();  
  allow read: if isUserBranch(resource.data.branch);  
  allow create: if isUserBranch(request.resource.data.branch);  
  allow update: if isUserBranch(resource.data.branch) && isUserBranch(request.resource.data.branch);  
}  
match /daily_attendance/{document=**} {  
  allow read, write, delete: if isAdmin();  
  allow read: if isUserBranch(resource.data.branch);  
  allow create: if isUserBranch(request.resource.data.branch);  
  allow update: if isUserBranch(resource.data.branch) && isUserBranch(request.resource.data.branch);  
}

}
}
